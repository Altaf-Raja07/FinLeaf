# Runtime dependencies for the FinLeaf ML service.
#
# The scientific stack is pinned by RANGE, not by exact version: these wheels
# are rebuilt per Python minor release, and an exact pin to a build that has
# not yet shipped a cp313 wheel fails the Docker build for a reason that has
# nothing to do with this code. The ranges below keep the API stable
# (FastAPI/Pydantic v2, scikit-learn v1) while allowing patch and minor
# upgrades inside a tested envelope.
#
# Targets python:3.13-slim (see Dockerfile); the code itself is 3.11+.

fastapi>=0.115,<1.0
uvicorn[standard]>=0.30,<1.0
pydantic>=2.7,<3.0

# numpy 2.3+ dropped cp311 wheels, so the floor is 1.26 to keep the service
# installable on 3.11 as well as 3.12/3.13.
numpy>=1.26,<3.0
pandas>=2.2,<4.0
scikit-learn>=1.5,<2.0
scipy>=1.11,<2.0
joblib>=1.3,<2.0
